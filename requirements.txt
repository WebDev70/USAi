# Runtime dependencies. Kept deliberately tiny (see docs/principles.md §1 —
# minimal, audited runtime surface). Everything else (coverage, bandit,
# pip-audit, Docker, make) is DEV/CI tooling, lives in requirements-dev.txt, and
# is NEVER listed here.
#
# Hash pins (supply-chain integrity): exact version + sha256 of the published
# artifacts. pip refuses to install if the bytes don't match. Because ANY --hash
# line turns on pip's --require-hashes mode for the whole file, every transitive
# dependency must also be pinned+hashed here — that is why typing_extensions
# appears below even though no first-party code imports it.
#
# Regenerate a hash after a version bump with:
#   pip download <pkg>==<ver> --no-deps -d /tmp/wheels && pip hash /tmp/wheels/<whl>
#
# ---------------------------------------------------------------------------
# python-dotenv — .env loading (backend/server.py).
#
# Version note + security posture:
#   GHSA-mf9w-mj56-hr94 (CVE-2026-28684) affects python-dotenv's set_key()/
#   unset_key() symlink handling, fixed in 1.2.2. This app ONLY ever calls
#   load_dotenv() (never set_key/unset_key), so the advisory is NOT exploitable
#   here. We do NOT pin >=1.2.2 because 1.2.2 requires Python >=3.10 and this
#   project supports the 3.9 baseline (CI tests 3.9 + 3.11). Instead,
#   scripts/security-scan.sh and CI carry a documented, reviewed ignore for this
#   specific advisory ID. Revisit when the 3.9 baseline is dropped.
python-dotenv==1.0.1 \
    --hash=sha256:f7b63ef50f1b690dddf550d03497b66d609393b40b564ed0d674909a68ebf16a \
    --hash=sha256:e324ee90a023d808f1959c46bcbc04446a10ced277783dc6ee09987c37ec10ca

# pypdf — PDF text extraction (backend/file_parser.py). Justified runtime dep:
# the PDF binary format cannot reasonably be parsed with the stdlib. Pure
# Python, no compiled extensions, single transitive dep (typing_extensions).
#
# DOCX deliberately does NOT get a library: .docx is a ZIP of XML, so
# backend/file_parser.py reads it with stdlib zipfile + xml.etree instead of
# python-docx, which would drag in the large, platform-specific lxml wheel.
#
# Version note + security posture:
#   6.21.4 (bumped 2026-11-07) clears the 18 GHSA advisories pip-audit flagged
#   against the previous 6.19.0 pin (fixed across 6.21.2–6.21.4). It is the
#   newest 6.x release; 7.0.0 requires Python >=3.10, which would break the 3.9
#   CI baseline. Hashes below are the published wheel and sdist, in that order.
pypdf==6.21.4 \
    --hash=sha256:33e4fb41ca352810eb4e9838aa0d82d6888a5f8bf682f43e99cce5faa666255c \
    --hash=sha256:c8817e3f888d47a087872ded69812c1ef30cacb871d6e053e0c00bbaf0bc2163

# typing_extensions — TRANSITIVE (required by pypdf), listed only because
# --require-hashes mode demands every installed package be pinned + hashed.
# 4.15.0 is the last release supporting the Python 3.9 CI baseline.
typing_extensions==4.15.0 \
    --hash=sha256:f0fa19c6845758ab08074a0cfa8b7aecb71c999ca73d62883bc25cc018c4e548 \
    --hash=sha256:0cea48d173cc12fa28ecabc3b837ea3cf6f38c6d1136f85cbaaf598984861466
